-- CLIENTES (Marketing/CRM)
CREATE TABLE IF NOT EXISTS clientes (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  nombre TEXT NOT NULL,
  contacto TEXT,
  email TEXT,
  telefono TEXT,
  estado TEXT CHECK (estado IN ('Activo', 'Inactivo')) DEFAULT 'Activo',
  industria TEXT,
  creado DATE DEFAULT CURRENT_DATE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- COTIZACIONES (Marketing)
CREATE TABLE IF NOT EXISTS cotizaciones (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  cliente_id BIGINT,
  proyecto_id BIGINT,
  titulo TEXT NOT NULL,
  monto NUMERIC(12,2) NOT NULL,
  estado TEXT CHECK (estado IN ('Pendiente', 'Aceptada', 'Rechazada')) DEFAULT 'Pendiente',
  fecha DATE,
  descripcion TEXT,
  validez TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- PROYECTOS (Gestión)
CREATE TABLE IF NOT EXISTS proyectos (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  nombre TEXT NOT NULL,
  cliente_id BIGINT,
  cotizacion_id BIGINT,
  estado TEXT CHECK (estado IN ('Pendiente', 'En Progreso', 'Completado')) DEFAULT 'Pendiente',
  prioridad TEXT CHECK (prioridad IN ('Alta', 'Media', 'Baja')) DEFAULT 'Media',
  inicio DATE,
  fin DATE,
  descripcion TEXT,
  monto NUMERIC(12,2),
  progreso INTEGER CHECK (progreso >= 0 AND progreso <= 100) DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- FINANZAS: INGRESOS
CREATE TABLE IF NOT EXISTS ingresos (

  created_at TIMESTAMPTZ DEFAULT NOW()

-- Función auto-asignar user_id
CREATE OR REPLACE FUNCTION set_user_id_on_insert()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.user_id IS NULL THEN
    NEW.user_id := auth.uid();
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 5. VISTAS (Actualizadas para respetar el contexto)
-- ============================================

CREATE OR REPLACE VIEW v_proyectos_completos AS
SELECT 
  p.*,
  c.nombre AS cliente_nombre,
  c.email AS cliente_email
FROM proyectos p
LEFT JOIN clientes c ON p.cliente_id = c.id;

CREATE OR REPLACE VIEW v_cotizaciones_completas AS
SELECT 
  q.*,
  c.nombre AS cliente_nombre,
  p.nombre AS proyecto_nombre
FROM cotizaciones q
LEFT JOIN clientes c ON q.cliente_id = c.id
LEFT JOIN proyectos p ON q.proyecto_id = p.id;

CREATE OR REPLACE VIEW v_empleados_stats AS
SELECT 
  e.*,
  COUNT(DISTINCT ap.proyecto_id) AS proyectos_asignados,
  COUNT(DISTINCT i.id) AS incidentes_registrados
FROM empleados e
LEFT JOIN asignaciones_proyecto ap ON e.id = ap.empleado_id AND ap.estado = 'Activo'
LEFT JOIN incidentes_ssoma i ON e.id = i.empleado_id
GROUP BY e.id;

CREATE OR REPLACE VIEW v_tareas_completas AS
SELECT 
  t.*,
  e.nombre AS empleado_nombre,
  e.cargo AS empleado_cargo,
  p.nombre AS proyecto_nombre
FROM tareas t
LEFT JOIN empleados e ON t.asignado_a = e.id
LEFT JOIN proyectos p ON t.proyecto_id = p.id;

-- ============================================
-- 6. TRIGGER AUTH: Crear perfil al registrarse
-- ============================================

CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, email, nombre_completo, rol)
  VALUES (
    NEW.id, 
    NEW.email, 
    COALESCE(NEW.raw_user_meta_data->>'nombre_completo', split_part(NEW.email, '@', 1)),
    COALESCE(NEW.raw_user_meta_data->>'rol', 'user')
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();
