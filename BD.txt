-- 🔹 1. CREAR TABLAS
-- ============================================

-- CLIENTES (CRM)
CREATE TABLE IF NOT EXISTS clientes (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  nombre TEXT NOT NULL,
  contacto TEXT,
  email TEXT,
  telefono TEXT,
  estado TEXT CHECK (estado IN ('Activo', 'Inactivo')) DEFAULT 'Activo',
  industria TEXT,
  creado DATE DEFAULT CURRENT_DATE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- COTIZACIONES (Marketing)
CREATE TABLE IF NOT EXISTS cotizaciones (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  cliente_id BIGINT,
  proyecto_id BIGINT,
  titulo TEXT NOT NULL,
  monto NUMERIC(12,2) NOT NULL,
  estado TEXT CHECK (estado IN ('Pendiente', 'Aceptada', 'Rechazada')) DEFAULT 'Pendiente',
  fecha DATE,
  descripcion TEXT,
  validez TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- PROYECTOS (Gestión)
CREATE TABLE IF NOT EXISTS proyectos (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  nombre TEXT NOT NULL,
  cliente_id BIGINT,
  cotizacion_id BIGINT,
  estado TEXT CHECK (estado IN ('Pendiente', 'En Progreso', 'Completado')) DEFAULT 'Pendiente',
  prioridad TEXT CHECK (prioridad IN ('Alta', 'Media', 'Baja')) DEFAULT 'Media',
  inicio DATE,
  fin DATE,
  descripcion TEXT,
  monto NUMERIC(12,2),
  progreso INTEGER CHECK (progreso >= 0 AND progreso <= 100) DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- FINANZAS: INGRESOS
CREATE TABLE IF NOT EXISTS ingresos (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  tipo TEXT CHECK (tipo IN ('Proyecto', 'Servicio', 'Otro')) DEFAULT 'Proyecto',
  concepto TEXT NOT NULL,
  monto NUMERIC(12,2) NOT NULL,
  fecha DATE,
  proyecto_id BIGINT,
  cliente_id BIGINT,
  estado TEXT CHECK (estado IN ('Pendiente', 'Cobrado')) DEFAULT 'Pendiente',
  metodo TEXT CHECK (metodo IN ('Transferencia', 'Efectivo', 'Tarjeta', 'Pendiente')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- FINANZAS: EGRESOS
CREATE TABLE IF NOT EXISTS egresos (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  categoria TEXT,
  concepto TEXT NOT NULL,
  monto NUMERIC(12,2) NOT NULL,
  fecha DATE,
  proyecto_id BIGINT,
  tipo TEXT CHECK (tipo IN ('Operativo', 'Proyecto')),
  metodo TEXT CHECK (metodo IN ('Tarjeta', 'Transferencia', 'Efectivo')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- FINANZAS: CUENTAS POR COBRAR
CREATE TABLE IF NOT EXISTS cuentas_por_cobrar (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  cliente_id BIGINT NOT NULL,
  proyecto_id BIGINT,
  concepto TEXT NOT NULL,
  monto NUMERIC(12,2) NOT NULL,
  fecha_emision DATE,
  fecha_vencimiento DATE,
  estado TEXT CHECK (estado IN ('Pendiente', 'Cobrada', 'Vencida')) DEFAULT 'Pendiente',
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- RRHH: EMPLEADOS
CREATE TABLE IF NOT EXISTS empleados (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  nombre TEXT NOT NULL,
  email TEXT UNIQUE,
  telefono TEXT,
  cargo TEXT,
  departamento TEXT,
  salario NUMERIC(10,2),
  fecha_ingreso DATE,
  estado TEXT CHECK (estado IN ('Activo', 'Inactivo')) DEFAULT 'Activo',
  tipo_contrato TEXT CHECK (tipo_contrato IN ('Indefinido', 'Temporal', 'Por Proyecto')),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- RRHH: ASISTENCIAS
CREATE TABLE IF NOT EXISTS asistencias (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  empleado_id BIGINT NOT NULL,
  fecha DATE NOT NULL,
  hora_entrada TIME,
  hora_salida TIME,
  estado TEXT CHECK (estado IN ('Presente', 'Tarde', 'Ausente', 'Vacaciones')) DEFAULT 'Presente',
  observaciones TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(empleado_id, fecha)
);

-- RRHH: ASIGNACIONES A PROYECTOS
CREATE TABLE IF NOT EXISTS asignaciones_proyecto (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  empleado_id BIGINT NOT NULL,
  proyecto_id BIGINT NOT NULL,
  rol TEXT,
  fecha_inicio DATE,
  fecha_fin DATE,
  estado TEXT CHECK (estado IN ('Activo', 'Completado', 'Pendiente')) DEFAULT 'Activo',
  horas_semanales INTEGER DEFAULT 40,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(empleado_id, proyecto_id)
);

-- RRHH: INCIDENTES SSOMA
CREATE TABLE IF NOT EXISTS incidentes_ssoma (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  empleado_id BIGINT NOT NULL,
  tipo TEXT CHECK (tipo IN ('Accidente Leve', 'Accidente Grave', 'Enfermedad Ocupacional', 'Casi Accidente')),
  descripcion TEXT,
  fecha DATE,
  gravedad TEXT CHECK (gravedad IN ('Baja', 'Media', 'Alta')),
  estado TEXT CHECK (estado IN ('Abierto', 'En Seguimiento', 'Cerrado')) DEFAULT 'Abierto',
  acciones_tomadas TEXT,
  fecha_reporte DATE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- RRHH: EXÁMENES MÉDICOS
CREATE TABLE IF NOT EXISTS examenes_medicos (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  empleado_id BIGINT NOT NULL,
  tipo TEXT CHECK (tipo IN ('Pre-ocupacional', 'Periódico', 'De Salida')),
  fecha DATE,
  resultado TEXT CHECK (resultado IN ('Apto', 'Apto con observaciones', 'No Apto')),
  observaciones TEXT,
  proximo_examen DATE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- LOGÍSTICA: ACTIVOS
CREATE TABLE IF NOT EXISTS activos (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  nombre TEXT NOT NULL,
  tipo TEXT CHECK (tipo IN ('Laptop', 'Monitor', 'Herramienta', 'Infraestructura', 'Periférico', 'Otros')),
  marca TEXT,
  modelo TEXT,
  serie TEXT UNIQUE,
  estado TEXT CHECK (estado IN ('Disponible', 'En uso', 'En mantenimiento', 'En tránsito', 'Dado de baja')) DEFAULT 'Disponible',
  fecha_compra DATE,
  ubicacion TEXT,
  costo NUMERIC(10,2),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- LOGÍSTICA: ASIGNACIONES DE ACTIVOS
CREATE TABLE IF NOT EXISTS asignaciones_activos (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  activo_id BIGINT NOT NULL,
  tipo_asignacion TEXT CHECK (tipo_asignacion IN ('Empleado', 'Proyecto')) NOT NULL,
  ref_id BIGINT NOT NULL,
  fecha_asignacion DATE,
  fecha_devolucion DATE,
  estado TEXT CHECK (estado IN ('Activa', 'Devuelta', 'Inactiva')) DEFAULT 'Activa',
  observaciones TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- LOGÍSTICA: MANTENIMIENTOS
CREATE TABLE IF NOT EXISTS mantenimientos (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  activo_id BIGINT NOT NULL,
  tipo TEXT CHECK (tipo IN ('Preventivo', 'Correctivo', 'Inspección')),
  descripcion TEXT,
  fecha DATE,
  costo NUMERIC(10,2),
  tecnico TEXT,
  estado TEXT CHECK (estado IN ('Pendiente', 'Completado')) DEFAULT 'Pendiente',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- LOGÍSTICA: GUÍAS DE SALIDA
CREATE TABLE IF NOT EXISTS guias_salida (
  id BIGINT PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
  activo_id BIGINT NOT NULL,
  destino TEXT NOT NULL,
  fecha_salida DATE,
  fecha_regreso DATE,
  estado TEXT CHECK (estado IN ('En tránsito', 'Devuelto', 'Pendiente')) DEFAULT 'En tránsito',
  responsable TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 🔹 3. TRIGGERS para updated_at
-- ============================================

CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_clientes_updated_at BEFORE UPDATE ON clientes FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_cotizaciones_updated_at BEFORE UPDATE ON cotizaciones FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_proyectos_updated_at BEFORE UPDATE ON proyectos FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_cuentas_por_cobrar_updated_at BEFORE UPDATE ON cuentas_por_cobrar FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_empleados_updated_at BEFORE UPDATE ON empleados FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_activos_updated_at BEFORE UPDATE ON activos FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- 🔹 4. VISTAS ÚTILES
-- ============================================

CREATE OR REPLACE VIEW v_proyectos_completos AS
SELECT 
  p.*,
  c.nombre AS cliente_nombre,
  c.email AS cliente_email
FROM proyectos p
LEFT JOIN clientes c ON p.cliente_id = c.id;

CREATE OR REPLACE VIEW v_cotizaciones_completas AS
SELECT 
  q.*,
  c.nombre AS cliente_nombre,
  p.nombre AS proyecto_nombre
FROM cotizaciones q
LEFT JOIN clientes c ON q.cliente_id = c.id
LEFT JOIN proyectos p ON q.proyecto_id = p.id;

CREATE OR REPLACE VIEW v_empleados_stats AS
SELECT 
  e.*,
  COUNT(DISTINCT ap.proyecto_id) AS proyectos_asignados,
  COUNT(DISTINCT i.id) AS incidentes_registrados
FROM empleados e
LEFT JOIN asignaciones_proyecto ap ON e.id = ap.empleado_id AND ap.estado = 'Activo'
LEFT JOIN incidentes_ssoma i ON e.id = i.empleado_id
GROUP BY e.id;
